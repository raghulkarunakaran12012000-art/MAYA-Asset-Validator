#______________________________________ASSET NAME CHECK_________________________________________#
 
def asset_name_check(asset_sm):
    meshname = []
    for i in asset_sm:
        asset_name = cmds.listRelatives(i, parent=True) # asset_name = []. so, in next line we use asset[0]
        if (asset_name[0].startswith("SM_") and re.search(r"\d+$", asset_name[0])):  
            meshname.append(asset_name[0] + " = perfect")
        else:
            meshname.append(asset_name[0] + " = Notperfect")
    print("___Mesh name___")
    return meshname

#______________________________________TRANSFORM CHECK_________________________________________#

def Transform_check(asset_sm):
    transform = []
    for i in asset_sm:
        pos = cmds.listRelatives(i,parent=True)[0]
        trans = cmds.xform(pos, query = True, translation=True)
        deg = cmds.getAttr(pos + ".rotate") 
        size = cmds.xform(pos, query = True, scale=True, worldSpace=True)
        if ((trans==[0.0, 0.0, 0.0]) and (deg==[(0.0,0.0,0.0)]) and (size==[1.0, 1.0, 1.0])):
	        transform.append((cmds.listRelatives(i, parent=True)[0]) + " = transform Perfect")
        else:
            if(trans!=[0.0, 0.0, 0.0]):
                transform.append(i + " = Moved")

            if(deg!=[(0.0,0.0,0.0)]):
                transform.append(i + " = Rotated")

            if(size!=[1.0, 1.0, 1.0]):
                transform.append(i + " = Scaled")
    print("___Transform check___")
    return transform

# Here we are checking Two things
	# Every faces are quad, Else it is not a quad
